"""Read-only MT5 target Demo probe; credentials remain in environment."""

import json
import os
import sys
sys.path.insert(0, os.environ["XAUUSD_REPO_ROOT"])
from datetime import datetime, timezone
from pathlib import Path


def main():
    import MetaTrader5 as mt5

    print(json.dumps({"stage": "import_ok", "version": mt5.__version__}), flush=True)
    login = int(os.environ["XAUUSD_MT5_LOGIN"])
    server = os.environ["XAUUSD_MT5_SERVER"]
    ok = mt5.initialize(
        path=os.getenv("MT5_TERMINAL_PATH", r"C:\Program Files\MetaTrader 5\terminal64.exe"),
        login=login,
        password=os.environ["XAUUSD_MT5_PASSWORD"],
        server=server,
        timeout=60000,
    )
    result = {
        "created_at_utc": datetime.now(timezone.utc).isoformat(),
        "initialized": bool(ok),
        "last_error": mt5.last_error(),
        "order_api_called": False,
    }
    try:
        if ok:
            a = mt5.account_info()
            verified = bool(
                a
                and a.login == login
                and a.server == server
                and a.trade_mode == mt5.ACCOUNT_TRADE_MODE_DEMO
            )
            result["target_demo_verified"] = verified
            if verified:
                result["account"] = {
                    k: getattr(a, k)
                    for k in (
                        "server",
                        "currency",
                        "balance",
                        "equity",
                        "margin_mode",
                        "leverage",
                        "margin_free",
                    )
                }
                si = mt5.symbol_info("XAUUSDm")
                if si:
                    result["symbol_spec"] = {
                        k: getattr(si, k)
                        for k in (
                            "name",
                            "trade_contract_size",
                            "trade_tick_size",
                            "trade_tick_value",
                            "trade_tick_value_profit",
                            "trade_tick_value_loss",
                            "volume_min",
                            "volume_step",
                            "volume_max",
                            "volume_limit",
                            "point",
                            "digits",
                            "trade_stops_level",
                            "trade_freeze_level",
                            "swap_long",
                            "swap_short",
                            "swap_mode",
                            "swap_rollover3days",
                            "trade_mode",
                            "trade_calc_mode",
                            "trade_exemode",
                            "filling_mode",
                            "currency_profit",
                            "currency_margin",
                        )
                    }
                tick = mt5.symbol_info_tick("XAUUSDm")
                if tick:
                    result["latest_tick"] = {
                        k: getattr(tick, k) for k in ("time", "time_msc", "bid", "ask")
                    }
        if result.get("target_demo_verified"):
            from scripts.pure_b1.broker_data import completed_m15_from_m1
            from scripts.strategy import trial30_bars
            from scripts.trial30.runner import MAGIC
            now = datetime.now(timezone.utc)
            begin = datetime(2026, 10, 1, tzinfo=timezone.utc)
            deals = mt5.history_deals_get(begin, now)
            orders = mt5.history_orders_get(begin, now)
            def summary(items):
                if items is None:
                    return {"available": False}
                selected = [x for x in items if x.symbol == "XAUUSDm"]
                own = [x for x in selected if str(x.comment).startswith("T30_")]
                return {"available": True, "symbol_count": len(selected), "trial30_prefix_count": len(own), "trial30_magic_count": sum(int(x.magic) == MAGIC for x in selected)}
            result["history_since_utc"] = begin.isoformat()
            result["deals"] = summary(deals)
            result["symbol_deal_times"] = [
                {"time_utc": datetime.fromtimestamp(int(x.time_msc)/1000, timezone.utc).isoformat(),
                 "type": int(x.type), "entry": int(x.entry), "volume": float(x.volume),
                 "trial30_magic": int(x.magic) == MAGIC}
                for x in (deals or ()) if x.symbol == "XAUUSDm"
            ]
            result["orders"] = summary(orders)
            rates = mt5.copy_rates_from_pos("XAUUSDm", mt5.TIMEFRAME_M1, 0, 5000)
            bars = trial30_bars(completed_m15_from_m1(rates, now=now))
            recent = bars.loc[bars.index >= begin]
            result["historical_signal_audit"] = {
                "available_completed_m15": len(recent),
                "sell_signals": int((recent.direction == -1).sum()),
                "sell_rows": [
                    {"close_utc": (stamp + __import__("pandas").Timedelta(minutes=15)).isoformat(),
                     "atr": float(row.atr),
                     "nominal_atr_stop_risk_usd": float(row.atr) * 2.5 * float(si.trade_contract_size) * 0.01}
                    for stamp, row in recent.loc[recent.direction == -1].iterrows()
                ],
                "last_bar_open_utc": bars.index[-1].isoformat(),
                "last_direction": int(bars.iloc[-1].direction),
                "note": "Historical detection only; not proof watcher observed or could admit these entries.",
            }
        dest = Path(os.environ["XAUUSD_PROBE_OUTPUT"])
        dest.parent.mkdir(parents=True, exist_ok=True)
        dest.write_text(json.dumps(result, indent=2))
        print(json.dumps(result), flush=True)
        return 0 if result.get("target_demo_verified") else 1
    finally:
        mt5.shutdown()


if __name__ == "__main__":
    raise SystemExit(main())
